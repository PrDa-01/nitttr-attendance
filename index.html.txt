<script>
  // 1. Read Class Details sent from the Flutter App URL
  const params = new URLSearchParams(window.location.search);
  const subject = params.get('subject') || 'Data Structures';
  const room = params.get('room') || 'Lecture Hall 1 (Block A)';
  const teacher = params.get('teacher') || 'Faculty Admin';

  document.getElementById('subject-display').innerText = "Subject: " + subject;
  document.getElementById('room-display').innerText = "Room: " + room;
  document.getElementById('teacher-display').innerText = "By: " + teacher;
  document.getElementById('date-display').innerText = new Date().toLocaleDateString();

  // 2. Student Roster (Enrollment No. & Name)
  const students = [
    { enrollmentId: "260101300", name: "Aarav Sharma", status: "Present" },
    { enrollmentId: "260101301", name: "Priya Patel", status: "Present" },
    { enrollmentId: "260101302", name: "Rohan Verma", status: "Present" },
    { enrollmentId: "260101303", name: "Sneha Gupta", status: "Present" },
    { enrollmentId: "260101304", name: "Vikram Singh", status: "Present" }
  ];

  function renderStudents() {
    const list = document.getElementById('student-list');
    list.innerHTML = '';
    students.forEach((student, index) => {
      list.innerHTML += `
        <div class="student-card">
          <div class="student-info">
            <span>Enrollment No: ${student.enrollmentId}</span>
            <h3>${student.name}</h3>
          </div>
          <div class="status-toggle">
            <button class="btn-status present ${student.status === 'Present' ? 'active' : ''}" 
                    onclick="setStatus(${index}, 'Present')">Present</button>
            <button class="btn-status absent ${student.status === 'Absent' ? 'active' : ''}" 
                    onclick="setStatus(${index}, 'Absent')">Absent</button>
          </div>
        </div>
      `;
    });
  }

  function setStatus(index, status) {
    students[index].status = status;
    renderStudents();
  }

  function confirmAttendance() {
    const presentCount = students.filter(s => s.status === 'Present').length;
    const msg = document.getElementById('success-msg');
    msg.style.display = 'block';
    msg.innerText = `Attendance Confirmed! (${presentCount}/${students.length} Present for ${subject})`;
  }

  renderStudents();
</script>
